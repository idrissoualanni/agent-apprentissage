**Goal:** Transformer l'agent d'apprentissage V1 en un système mature avec human-in-the-loop, RAG conditionnel, outils enrichis, interface interactive, et support de modèles cloud.

**Architecture:** Le graphe LangGraph existant est étendu avec des nœuds de confirmation (human-in-the-loop), un router conditionnel pour le RAG, des outils web search + revision planner, et des responses HTML interactives (quiz, Feynman, artefacts). """Outil de recherche web pour enrichir les réponses."""

import json
from langchain.tools import tool
from duckduckgo_search import DDGS


@tool
def web_search(query: str, num_results: int = 3) -> str:
    """Recherche sur le web pour enrichir une réponse.

    Args:
        query: Requête de recherche
        num_results: Nombre de résultats (défaut: 3)

    Returns:
        JSON string avec les résultats (title, url, snippet)
    """
    try:
        with DDGS() as ddgs:
            results = list(ddgs.text(query, max_results=num_results))
            formatted = [
                {"title": r.get("title", ""), "url": r.get("href", ""), "snippet": r.get("body", "")}
                for r in results
            ]
            return json.dumps(formatted, ensure_ascii=False)
    except Exception as e:
        return json.dumps({"error": str(e)})
```

@tool
def get_revision_plan(domain: str) -> str:
    """Génère un plan de révision basé sur les compétences en retard.

    Args:
        domain: Domaine d'apprentissage

    Returns:
        JSON string avec plan de révision ordonné
    """
    due = db.get_due_for_review(config.DB_PATH)
    if not due:
        return json.dumps({"plan": [], "message": "Aucune révision nécessaire aujourd'hui."})

    # Trier par urgence (plus ancien d'abord)
    plan = [
        {
            "competency_id": c["id"],
            "nom": c["nom"],
            "score": c["score"],
            "leitner_box": c["leitner_box"],
            "next_review": c["next_review_at"],
        }
        for c in due[:5]  # Max 5 par session
    ]

    return json.dumps({
        "plan": plan,
        "total_due": len(due),
        "message": f"{len(plan)} compétence(s) à réviser.",
    }, ensure_ascii=False)
```

- [ ] **Step 2: Ajouter le nœud revision_planner dans nodes.py**

def revision_planner_node(state: dict, db_path=None) -> dict:
    """Vérifie les révisions dues et propose un plan."""
    profile = db.get_profile(db_path)
    domain = profile.get("domain", "")
    if not domain:
        return {}

    due = db.get_due_for_review(db_path)
    if not due:
        return {}

    return {
        "method": "revision",
        "tool_name": "revision_plan",
        "tool_result": json.dumps([{"id": c["id"], "nom": c["nom"]} for c in due[:5]]),
    }

"""Fournisseurs de modèles cloud."""

import os
from langchain_openai import ChatOpenAI
from langchain_anthropic import ChatAnthropic


def get_openai_model(model_name: str = "gpt-4o-mini"):
    api_key = os.environ.get("OPENAI_API_KEY")
    if not api_key:
        raise ValueError("OPENAI_API_KEY not set")
    return ChatOpenAI(model=model_name, api_key=api_key)


def get_anthropic_model(model_name: str = "claude-3-5-sonnet-20241022"):
    api_key = os.environ.get("ANTHROPIC_API_KEY")
    if not api_key:
        raise ValueError("ANTHROPIC_API_KEY not set")
    return ChatAnthropic(model=model_name, api_key=api_key)
